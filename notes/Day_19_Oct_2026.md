#program:

import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
    public static void main(String[] args) throws java.lang.Exception
    {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        HashSet < Integer > set = new HashSet < > ();
        int ind = 0;
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
            if (!set.contains(arr[i])) {
                set.add(arr[i]);
                arr[ind++] = arr[i];
            }
        }
        for (int i = 0; i < ind; i++) {
            System.out.print(arr[i] + " ");
        }
    }
}
