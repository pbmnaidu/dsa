import java.util.Scanner;

class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		for (int i = 1; i <= n; i++) {
			for (int j = 0; j < n - i; j++) {
				System.out.print(" ");
			}
			int s = 2*i - 1;
			for (int j = 0; j<s; j++) {
				System.out.print("*");
			}
			System.out.println(" ");
		}
		sc.close();
	}
}
