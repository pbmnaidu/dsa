import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String r = (n%2 == 0 ) ? "even" : "odd" ;
        System.out.println(r);
    }
}
