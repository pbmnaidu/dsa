import java.util.Scanner;

class Main {
    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int n = sc.nextInt();
        float f = sc.nextFloat();
        int ff = (int)f;
        System.out.println(s);
        System.out.println(n);
        System.out.println(ff);
    }
}
